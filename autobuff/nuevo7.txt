<script>
    // Leer archivos locales y ponerlos en los textareas
    function handleFileSelect(evt, textAreaId) {
        const file = evt.target.files[0];
        if (!file) return;

        const reader = new FileReader();
        reader.onload = function(e) {
            document.getElementById(textAreaId).value = e.target.result;
        };
        reader.readAsText(file);
    }

    document.getElementById('fileA').addEventListener('change', (e) => handleFileSelect(e, 'textA'));
    document.getElementById('fileB').addEventListener('change', (e) => handleFileSelect(e, 'textB'));

    // Algoritmo básico de comparación (Por palabras)
    document.getElementById('compareBtn').addEventListener('click', () => {
        const textA = document.getElementById('textA').value;
        const textB = document.getElementById('textB').value;
        
        const outputDiv = document.getElementById('diffResult');
        const resultSection = document.getElementById('resultSection');

        if (!textA && !textB) {
            alert("Por favor, ingresa texto en ambos campos o sube archivos.");
            return;
        }

        // Dividir los textos en palabras conservando espacios y saltos de línea
        const wordsA = textA.split(/(\s+)/);
        const wordsB = textB.split(/(\s+)/);

        // Mostrar sección de resultados
        resultSection.style.display = 'block';
        outputDiv.innerHTML = '';

        // Objeto para trackear las posiciones
        let i = 0;
        let j = 0;
        let fragment = document.createDocumentFragment();

        // Algoritmo de comparación simple lineal
        while (i < wordsA.length || j < wordsB.length) {
            if (i < wordsA.length && j < wordsB.length && wordsA[i] === wordsB[j]) {
                // Las palabras coinciden
                let span = document.createElement('span');
                span.textContent = wordsA[i];
                fragment.appendChild(span);
                i++;
                j++;
            } else {
                // Buscar si la palabra de B aparece más adelante en A (Eliminación)
                let lookAheadA = wordsA.indexOf(wordsB[j], i);
                // Buscar si la palabra de A aparece más adelante en B (Inserción)
                let lookAheadB = wordsB.indexOf(wordsA[i], j);

                if (lookAheadA !== -1 && (lookAheadB === -1 || lookAheadA - i < lookAheadB - j)) {
                    // Se eliminó contenido de A
                    while (i < lookAheadA) {
                        if(wordsA[i].trim() !== "") {
                            let del = document.createElement('del');
                            del.textContent = wordsA[i];
                            fragment.appendChild(del);
                        } else {
                            fragment.appendChild(document.createTextNode(wordsA[i]));
                        }
                        i++;
                    }
                } else if (lookAheadB !== -1) {
                    // Se insertó contenido en B
                    while (j < lookAheadB) {
                        if(wordsB[j].trim() !== "") {
                            let ins = document.createElement('ins');
                            ins.textContent = wordsB[j];
                            fragment.appendChild(ins);
                        } else {
                            fragment.appendChild(document.createTextNode(wordsB[j]));
                        }
                        j++;
                    }
                } else {
                    // Son completamente diferentes en este punto (Reemplazo)
                    if (i < wordsA.length) {
                        if(wordsA[i].trim() !== "") {
                            let del = document.createElement('del');
                            del.textContent = wordsA[i];
                            fragment.appendChild(del);
                        } else {
                            fragment.appendChild(document.createTextNode(wordsA[i]));
                        }
                        i++;
                    }
                    if (j < wordsB.length) {
                        if(wordsB[j].trim() !== "") {
                            let ins = document.createElement('ins');
                            ins.textContent = wordsB[j];
                            fragment.appendChild(ins);
                        } else {
                            fragment.appendChild(document.createTextNode(wordsB[j]));
                        }
                        j++;
                    }
                }
            }
        }
        outputDiv.appendChild(fragment);
    });
</script>
